      showFormMessage(
        "error",
        "Website chưa được kết nối dịch vụ email. Cần thêm endpoint thật trước khi gửi."
      );
      return;
    }

    /*
      Tạo form gửi dữ liệu trực tiếp.
      Trình duyệt sẽ chuyển đến dịch vụ nhận form.
    */
    const sendForm = document.createElement("form");
    sendForm.method = "POST";
    sendForm.action = emailEndpoint;
    sendForm.enctype = "multipart/form-data";

    const fields = new FormData(joinForm);

    fields.forEach(function (value, key) {
      if (typeof value === "string") {
        const input = document.createElement("input");
        input.type = "hidden";
        input.name = key;
        input.value = value;
        sendForm.appendChild(input);
      }
    });

    document.body.appendChild(sendForm);

    showFormMessage(
      "success",
      "Đang chuyển đơn đăng ký đến dịch vụ nhận form. Hãy chờ kết quả xác nhận."
    );

    sendForm.submit();
  });

  function showFormMessage(type, message) {
    formMessage.className = "form-message " + type;
    formMessage.textContent = message;

    if (message) {
      formMessage.scrollIntoView({
        behavior: "smooth",
        block: "nearest"
      });
    }
  }
</script>
